# ---- 1. One-time server setup ----
sudo apt update
sudo apt upgrade -y
sudo apt install -y git curl unzip tar make sudo vim wget python3-pip python3-venv python3-full

# ---- 2. Get the code (already done, so skip this) ----
# git clone https://github.com/shushant11/Resume-Genie.git
cd ~/Resume-Genie

# ---- 3. Create and activate a virtual environment ----
python3 -m venv .venv
source .venv/bin/activate

# ---- 4. Install dependencies (inside the venv) ----
pip install -r requirements.txt

# ---- 5a. Temporary run (stops when you close the terminal) ----
python -m streamlit run app.py --server.address 0.0.0.0 --server.port 8501

# ---- 5b. Permanent run (keeps running after you disconnect) ----
nohup .venv/bin/python -m streamlit run app.py --server.address 0.0.0.0 --server.port 8501 > app.log 2>&1 &

# Useful commands
tail -f app.log          # watch the logs (Ctrl+C stops watching, not the app)
pgrep -af streamlit      # check the app is running
pkill -f "streamlit run" # stop the app
